## 3. Core Algorithms

### 3.1 Span Extraction Algorithm

```rust
// avocado-core/src/span.rs

/// Extract spans from document text
/// Rules:
/// - Target 20-50 lines per span
/// - Respect paragraph boundaries (double newlines)
/// - Keep code blocks together
/// - Never split mid-sentence
pub fn extract_spans(content: &str) -> Vec<Span> {
    let lines: Vec<&str> = content.lines().collect();
    let mut spans = Vec::new();
    let mut current_start = 1;
    let mut current_lines = Vec::new();
    
    for (idx, line) in lines.iter().enumerate() {
        let line_num = idx + 1;
        current_lines.push(*line);
        
        // Check if we should create a span
        let should_split = 
            // Reached target size
            current_lines.len() >= 30 ||
            // Hit paragraph boundary and have minimum size
            (line.trim().is_empty() && current_lines.len() >= 20) ||
            // End of document
            idx == lines.len() - 1;
        
        if should_split && !current_lines.is_empty() {
            let text = current_lines.join("\n");
            let token_count = estimate_tokens(&text);
            
            spans.push(Span {
                id: generate_uuid(),
                artifact_id: String::new(), // Set during ingestion
                start_line: current_start,
                end_line: line_num,
                text,
                embedding: None,
                embedding_model: None,
                token_count,
                metadata: None,
            });
            
            current_start = line_num + 1;
            current_lines.clear();
        }
    }
    
    spans
}

fn estimate_tokens(text: &str) -> usize {
    // Rough estimate: ~4 chars per token
    // Use tiktoken-rs for accurate counts
    text.len() / 4
}
```

### 3.3 Context Compilation Algorithm

```rust
// avocado-core/src/compiler.rs

pub async fn compile(
    query: &str, 
    config: CompilerConfig,
    db: &Database,
    index: &VectorIndex,
) -> Result<WorkingSet> {
    let start_time = std::time::Instant::now();
    
    // Step 1: Embed query
    let query_embedding = embed_text(query, &api_key).await?;
    
    // Step 2: Semantic search
    let semantic_results = index.search(&query_embedding, 50)?;
    
    // Step 3: Lexical search (simple keyword matching)
    let lexical_results = lexical_search(query, db, 20)?;
    
    // Step 4: Hybrid fusion
    let mut candidates = hybrid_fusion(
        semantic_results,
        lexical_results,
        config.semantic_weight,
        config.lexical_weight,
    );
    
    // Step 5: MMR diversification
    if config.enable_mmr {
        candidates = apply_mmr(candidates, &query_embedding, config.mmr_lambda);
    }
    
    // Step 6: Pack into token budget
    let selected_spans = pack_token_budget(candidates, config.token_budget);
    
    // Step 7: Sort deterministically
    let sorted_spans = deterministic_sort(selected_spans);
    
    // Step 8: Build context and citations
    let (context_text, citations) = build_context(sorted_spans);
    
    Ok(WorkingSet {
        text: context_text,
        spans: sorted_spans,
        citations,
        tokens_used: count_tokens(&context_text),
        query: query.to_string(),
        compilation_time_ms: start_time.elapsed().as_millis() as u64,
    })
}

fn hybrid_fusion(
    semantic: Vec<ScoredSpan>,
    lexical: Vec<ScoredSpan>,
    semantic_weight: f32,
    lexical_weight: f32,
) -> Vec<ScoredSpan> {
    // Reciprocal Rank Fusion
    let mut scores: HashMap<String, f32> = HashMap::new();
    
    // Add semantic scores
    for (rank, span) in semantic.into_iter().enumerate() {
        let score = semantic_weight / (60.0 + rank as f32);
        scores.insert(span.id.clone(), score);
    }
    
    // Add lexical scores
    for (rank, span) in lexical.into_iter().enumerate() {
        let score = lexical_weight / (60.0 + rank as f32);
        *scores.entry(span.id.clone()).or_insert(0.0) += score;
    }
    
    // Convert back to sorted list
    let mut results: Vec<ScoredSpan> = scores
        .into_iter()
        .map(|(id, score)| ScoredSpan { id, score })
        .collect();
    
    results.sort_by(|a, b| b.score.partial_cmp(&a.score).unwrap());
    results
}

fn apply_mmr(
    candidates: Vec<ScoredSpan>,
    query_embedding: &[f32],
    lambda: f32,
) -> Vec<ScoredSpan> {
    let mut selected = Vec::new();
    let mut remaining = candidates;
    
    // Select first span (highest relevance)
    if let Some(first) = remaining.pop() {
        selected.push(first);
    }
    
    // Iteratively select diverse spans
    while !remaining.is_empty() && selected.len() < 30 {
        let mut best_score = f32::NEG_INFINITY;
        let mut best_idx = 0;
        
        for (idx, candidate) in remaining.iter().enumerate() {
            // Relevance to query
            let relevance = candidate.score;
            
            // Maximum similarity to already selected
            let max_sim = selected
                .iter()
                .map(|s| cosine_similarity(&candidate.embedding, &s.embedding))
                .max_by(|a, b| a.partial_cmp(b).unwrap())
                .unwrap_or(0.0);
            
            // MMR score
            let mmr_score = lambda * relevance - (1.0 - lambda) * max_sim;
            
            if mmr_score > best_score {
                best_score = mmr_score;
                best_idx = idx;
            }
        }
        
        selected.push(remaining.swap_remove(best_idx));
    }
    
    selected
}

fn pack_token_budget(
    mut candidates: Vec<ScoredSpan>,
    budget: usize,
) -> Vec<Span> {
    // Greedy packing - good enough for v1
    let mut selected = Vec::new();
    let mut total_tokens = 0;
    
    for candidate in candidates {
        if total_tokens + candidate.token_count <= budget {
            total_tokens += candidate.token_count;
            selected.push(candidate);
        }
    }
    
    selected
}

fn deterministic_sort(mut spans: Vec<Span>) -> Vec<Span> {
    // Sort by (artifact_id, start_line) for deterministic ordering
    spans.sort_by_key(|s| (s.artifact_id.clone(), s.start_line));
    spans
}
```

### 4.3 Python SDK API

```python
# avocado.py - Public API

from typing import List, Dict, Optional
import requests

class AvocadoDB:
    """Client for AvocadoDB context engine."""
    
    def __init__(self, url: str = "http://localhost:8080"):
        """Initialize client.
        
        Args:
            url: Server URL (default: http://localhost:8080)
        """
        self.url = url
    
    def ingest(self, path: str, content: Optional[str] = None) -> Dict:
        """Ingest a document.
        
        Args:
            path: Document path or identifier (must be unique)
            content: Document text. If None, reads from filesystem path
            
        Returns:
            Dict with artifact_id, spans_created
        """
        if content is None:
            with open(path, 'r') as f:
                content = f.read()
        
        response = requests.post(
            f"{self.url}/ingest",
            json={"path": path, "content": content}
        )
        return response.json()
    
    def compile(
        self,
        query: str,
        budget: int = 8000,
        config: Optional[Dict] = None
    ) -> 'WorkingSet':
        """Compile a context working set.
        
        Args:
            query: Search query
            budget: Token budget (default: 8000)
            config: Optional compiler configuration
            
        Returns:
            WorkingSet object with compiled context
        """
        payload = {"query": query, "token_budget": budget}
        if config:
            payload["config"] = config
            
        response = requests.post(f"{self.url}/compile", json=payload)
        return WorkingSet(response.json()["working_set"])
    
    def stats(self) -> Dict:
        """Get database statistics."""
        response = requests.get(f"{self.url}/stats")
        return response.json()
    
    def clear(self) -> None:
        """Clear all data."""
        requests.delete(f"{self.url}/clear")


class WorkingSet:
    """A compiled context working set."""
    
    def __init__(self, data: Dict):
        self.text = data["text"]
        self.spans = data["spans"]
        self.citations = data["citations"]
        self.tokens_used = data["tokens_used"]
        self.query = data.get("query", "")
        self.compilation_time_ms = data.get("compilation_time_ms", 0)
    
    def get_citation(self, text_fragment: str) -> Optional[Dict]:
        """Get citation for a text fragment."""
        # Match fragment to span and return citation
        for citation in self.citations:
            if text_fragment in citation["text"]:
                return citation
        return None
    
    def explain(self) -> Dict:
        """Get explanation of compilation process."""
        return {
            "query": self.query,
            "spans_included": len(self.spans),
            "tokens_used": self.tokens_used,
            "compilation_time_ms": self.compilation_time_ms,
            "citations": len(self.citations)
        }
```

### 5.1 Database Operations

```rust
// avocado-core/src/db.rs

use rusqlite::{Connection, Result};
use std::sync::Arc;
use parking_lot::RwLock;

pub struct Database {
    conn: Arc<RwLock<Connection>>,
}

impl Database {
    pub fn new(path: &str) -> Result<Self> {
        let conn = Connection::open(path)?;
        
        // Enable foreign keys
        conn.execute("PRAGMA foreign_keys = ON", [])?;
        
        // Run migrations
        conn.execute_batch(include_str!("../../migrations/001_initial.sql"))?;
        
        Ok(Self {
            conn: Arc::new(RwLock::new(conn)),
        })
    }
    
    pub fn insert_artifact(&self, artifact: &Artifact) -> Result<()> {
        let conn = self.conn.write();
        conn.execute(
            "INSERT INTO artifacts (id, path, content, content_hash, metadata, created_at)
             VALUES (?1, ?2, ?3, ?4, ?5, ?6)",
            params![
                artifact.id,
                artifact.path,
                artifact.content,
                artifact.content_hash,
                artifact.metadata.as_ref().map(|m| m.to_string()),
                artifact.created_at,
            ],
        )?;
        Ok(())
    }
    
    pub fn insert_spans(&self, spans: &[Span]) -> Result<()> {
        let conn = self.conn.write();
        let tx = conn.transaction()?;
        
        for span in spans {
            tx.execute(
                "INSERT INTO spans (
                    id, artifact_id, start_line, end_line, text,
                    embedding, embedding_model, token_count, metadata
                ) VALUES (?1, ?2, ?3, ?4, ?5, ?6, ?7, ?8, ?9)",
                params![
                    span.id,
                    span.artifact_id,
                    span.start_line,
                    span.end_line,
                    span.text,
                    span.embedding.as_ref().map(|e| serialize_embedding(e)),
                    span.embedding_model,
                    span.token_count,
                    span.metadata.as_ref().map(|m| m.to_string()),
                ],
            )?;
        }
        
        tx.commit()?;
        Ok(())
    }
    
    pub fn get_all_spans(&self) -> Result<Vec<Span>> {
        let conn = self.conn.read();
        let mut stmt = conn.prepare(
            "SELECT id, artifact_id, start_line, end_line, text, 
                    embedding, embedding_model, token_count, metadata
             FROM spans"
        )?;
        
        let spans = stmt.query_map([], |row| {
            Ok(Span {
                id: row.get(0)?,
                artifact_id: row.get(1)?,
                start_line: row.get(2)?,
                end_line: row.get(3)?,
                text: row.get(4)?,
                embedding: row.get::<_, Option<Vec<u8>>>(5)?
                    .map(|bytes| deserialize_embedding(&bytes)),
                embedding_model: row.get(6)?,
                token_count: row.get(7)?,
                metadata: row.get::<_, Option<String>>(8)?
                    .and_then(|s| serde_json::from_str(&s).ok()),
            })
        })?
        .collect::<Result<Vec<_>>>()?;
        
        Ok(spans)
    }
}

fn serialize_embedding(embedding: &[f32]) -> Vec<u8> {
    // Convert f32 array to bytes for storage
    embedding.iter()
        .flat_map(|f| f.to_le_bytes())
        .collect()
}

fn deserialize_embedding(bytes: &[u8]) -> Vec<f32> {
    // Convert bytes back to f32 array
    bytes.chunks_exact(4)
        .map(|chunk| f32::from_le_bytes([chunk[0], chunk[1], chunk[2], chunk[3]]))
        .collect()
}
```

### 5.2 Vector Index Implementation

```rust
// avocado-core/src/index.rs

use std::collections::HashMap;

pub struct VectorIndex {
    spans: Vec<IndexedSpan>,
    embeddings: Vec<Vec<f32>>,
}

struct IndexedSpan {
    id: String,
    artifact_id: String,
    embedding_idx: usize,
}

impl VectorIndex {
    pub fn build(spans: Vec<Span>) -> Self {
        let mut indexed_spans = Vec::new();
        let mut embeddings = Vec::new();
        
        for span in spans {
            if let Some(embedding) = span.embedding {
                let idx = embeddings.len();
                embeddings.push(embedding);
                indexed_spans.push(IndexedSpan {
                    id: span.id,
                    artifact_id: span.artifact_id,
                    embedding_idx: idx,
                });
            }
        }
        
        Self {
            spans: indexed_spans,
            embeddings,
        }
    }
    
    pub fn search(&self, query: &[f32], k: usize) -> Vec<ScoredSpan> {
        // Brute force for Phase 1 (works fine for < 10K spans)
        let mut scores: Vec<(usize, f32)> = self.embeddings
            .iter()
            .enumerate()
            .map(|(idx, emb)| (idx, cosine_similarity(query, emb)))
            .collect();
        
        // Sort by score descending
        scores.sort_by(|a, b| b.1.partial_cmp(&a.1).unwrap());
        
        // Take top k
        scores.truncate(k);
        
        // Convert to ScoredSpan
        scores.into_iter()
            .filter_map(|(idx, score)| {
                self.spans.iter()
                    .find(|s| s.embedding_idx == idx)
                    .map(|s| ScoredSpan {
                        id: s.id.clone(),
                        artifact_id: s.artifact_id.clone(),
                        score,
                    })
            })
            .collect()
    }
}

fn cosine_similarity(a: &[f32], b: &[f32]) -> f32 {
    let dot: f32 = a.iter().zip(b.iter()).map(|(x, y)| x * y).sum();
    let norm_a: f32 = a.iter().map(|x| x * x).sum::<f32>().sqrt();
    let norm_b: f32 = b.iter().map(|x| x * x).sum::<f32>().sqrt();
    dot / (norm_a * norm_b)
}
```

```python
# langchain_integration.py - Drop-in replacement example

# Before: Using LangChain with standard vector store
from langchain.vectorstores import FAISS
from langchain.embeddings import OpenAIEmbeddings

def old_way(query, documents):
    vectorstore = FAISS.from_documents(documents, OpenAIEmbeddings())
    retriever = vectorstore.as_retriever(k=10)
    docs = retriever.get_relevant_documents(query)
    context = "\n\n".join([d.page_content for d in docs])
    return context  # Non-deterministic, no citations

# After: Using AvocadoDB
from avocado import AvocadoDB

def new_way(query, documents):
    db = AvocadoDB()
    
    # One-time ingestion
    for doc in documents:
        db.ingest(doc.metadata["source"], doc.page_content)
    
    # Deterministic compilation
    result = db.compile(query, budget=8000)
    return result.text  # Deterministic, with citations

# The integration is that simple!
```

## Appendix B: Troubleshooting Guide

### Common Issues

1. **Embeddings taking too long**
   - Batch requests (up to 2048 per call)
   - Implement caching aggressively
   - Consider ada-002 (fastest model)

2. **Non-deterministic results**
   - Check floating point comparisons
   - Ensure stable sorting with tie-breakers
   - Verify no randomness in algorithms

3. **Token counts incorrect**
   - Use tiktoken-rs for accurate counts
   - Account for special tokens
   - Test with actual OpenAI tokenizer

4. **Database locked errors**
   - Use connection pooling
   - Implement proper transaction management
   - Consider WAL mode for SQLite
